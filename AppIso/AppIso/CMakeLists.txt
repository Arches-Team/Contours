cmake_minimum_required(VERSION 3.21)

project(AppIso LANGUAGES CXX)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# moc runs automatically on any Q_OBJECT header listed among the target's
# sources below (main-window.h, iso-view.h, LibMaya/Include/realtime.h).
set(CMAKE_AUTOMOC ON)

find_package(Qt6 REQUIRED COMPONENTS Core Gui Widgets OpenGL OpenGLWidgets Svg)
find_package(OpenGL REQUIRED)
find_package(OpenMP REQUIRED)

# --- GLEW -----------------------------------------------------------------
# On Windows we reuse the GLEW binaries vendored in LibGL/ (same ones the
# Visual Studio project links against). On Linux, install GLEW from your
# package manager (e.g. `sudo apt install libglew-dev`) and find_package
# will pick it up.
if(WIN32)
  add_library(AppIso::GLEW SHARED IMPORTED)
  set_target_properties(AppIso::GLEW PROPERTIES
    IMPORTED_LOCATION "${CMAKE_CURRENT_SOURCE_DIR}/LibGL/bin/glew32.dll"
    IMPORTED_IMPLIB "${CMAKE_CURRENT_SOURCE_DIR}/LibGL/Library/glew32.lib"
    INTERFACE_INCLUDE_DIRECTORIES "${CMAKE_CURRENT_SOURCE_DIR}/LibGL/Include"
  )
else()
  find_package(GLEW REQUIRED)
  add_library(AppIso::GLEW ALIAS GLEW::GLEW)
endif()

# --- uic --------------------------------------------------------------
# Generated by hand (instead of CMAKE_AUTOUIC) so the output keeps the
# "uic_interface.h" name that Include/main-window.h expects, matching what
# the Visual Studio build already produces.
set(UIC_HEADER "${CMAKE_CURRENT_SOURCE_DIR}/Include/uic_interface.h")
add_custom_command(
  OUTPUT "${UIC_HEADER}"
  COMMAND Qt6::uic "${CMAKE_CURRENT_SOURCE_DIR}/UI/interface.ui" -o "${UIC_HEADER}"
  DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/UI/interface.ui"
  COMMENT "uic: interface.ui -> Include/uic_interface.h"
  VERBATIM
)
add_custom_target(AppIso_uic DEPENDS "${UIC_HEADER}")

if(WIN32)
  target_link_libraries(AppIso PRIVATE glu32)

  # Deploy the Qt runtime DLLs and the vendored glew32.dll next to the
  # executable, the same way the Visual Studio build's post-build step does.
  get_target_property(_qt_qmake_executable Qt6::qmake IMPORTED_LOCATION)
  get_filename_component(_qt_bin_dir "${_qt_qmake_executable}" DIRECTORY)

  add_custom_command(TARGET AppIso POST_BUILD
    COMMAND "${_qt_bin_dir}/windeployqt.exe" $<TARGET_FILE:AppIso>
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
            "${CMAKE_CURRENT_SOURCE_DIR}/LibGL/bin/glew32.dll"
            "$<TARGET_FILE_DIR:AppIso>"
    COMMENT "Deploying Qt runtime and glew32.dll next to AppIso.exe"
  )

  # main-window.h resolves its default Data/dems and Images/input directories
  # relative to the executable's own folder, so Data/ and Images/ (which live
  # next to this CMakeLists.txt's parent directory) need to be reachable from
  # there too. Use a directory junction rather than copying so the (possibly
  # large) DEM/image files aren't duplicated on disk and stay in sync.
  add_custom_command(TARGET AppIso POST_BUILD
    COMMAND "${CMAKE_CURRENT_SOURCE_DIR}/link-resource.bat" "$<TARGET_FILE_DIR:AppIso>/Data" "${CMAKE_CURRENT_SOURCE_DIR}/../Data"
    COMMAND "${CMAKE_CURRENT_SOURCE_DIR}/link-resource.bat" "$<TARGET_FILE_DIR:AppIso>/Images" "${CMAKE_CURRENT_SOURCE_DIR}/../Images"
    COMMENT "Linking Data/ and Images/ next to AppIso.exe"
    VERBATIM
  )
endif()
